cmake_minimum_required(VERSION 3.22)
project(BrushkinExport LANGUAGES C CXX)
include(FetchContent)
include(ExternalProject)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_RELEASE "-Os -DNDEBUG -ffunction-sections -fdata-sections")
set(CMAKE_CXX_FLAGS_RELEASE "-Os -DNDEBUG -ffunction-sections -fdata-sections")
FetchContent_Declare(jpeg URL https://github.com/libjpeg-turbo/libjpeg-turbo/releases/download/3.1.2/libjpeg-turbo-3.1.2.tar.gz URL_HASH SHA256=8f0012234b464ce50890c490f18194f913a7b1f4e6a03d6644179fa0f867d0cf)
FetchContent_Populate(jpeg)
ExternalProject_Add(jpeg_build SOURCE_DIR ${jpeg_SOURCE_DIR} BINARY_DIR ${jpeg_BINARY_DIR}
 CMAKE_ARGS -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM} -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE} -DANDROID_ABI=${ANDROID_ABI} -DANDROID_PLATFORM=android-29 -DCMAKE_BUILD_TYPE=Release -DENABLE_SHARED=OFF -DWITH_TURBOJPEG=OFF -DWITH_SIMD=OFF -DCMAKE_POSITION_INDEPENDENT_CODE=ON "-DCMAKE_C_FLAGS_RELEASE=-Os -DNDEBUG -ffunction-sections -fdata-sections"
 BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --target jpeg-static --parallel 4
 INSTALL_COMMAND "" BUILD_BYPRODUCTS ${jpeg_BINARY_DIR}/libjpeg.a)
add_library(jpeg_codec STATIC IMPORTED)
set_target_properties(jpeg_codec PROPERTIES IMPORTED_LOCATION ${jpeg_BINARY_DIR}/libjpeg.a)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(PNG_EXECUTABLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(png URL https://download.sourceforge.net/libpng/libpng-1.6.50.tar.gz URL_HASH SHA256=708f4398f996325819936d447f982e0db90b6b8212b7507e7672ea232210949a)
FetchContent_MakeAvailable(png)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(tool ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO WEBPMUX EXTRAS)
 set(WEBP_BUILD_${tool} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_BUILD_LIBWEBPMUX ON CACHE BOOL "" FORCE)
FetchContent_Declare(webp URL https://storage.googleapis.com/downloads.webmproject.org/releases/webp/libwebp-1.6.0.tar.gz URL_HASH SHA256=e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564)
FetchContent_MakeAvailable(webp)
FetchContent_Declare(gif URL https://downloads.sourceforge.net/project/giflib/giflib-6.x/giflib-6.1.3.tar.gz URL_HASH SHA256=b65b66b99f0424b93525f987386f22fc5efb9da2bfc92ad4a532249aaffbab0e)
FetchContent_Populate(gif)
add_library(gif_codec STATIC ${gif_SOURCE_DIR}/dgif_lib.c ${gif_SOURCE_DIR}/egif_lib.c ${gif_SOURCE_DIR}/gifalloc.c ${gif_SOURCE_DIR}/gif_err.c ${gif_SOURCE_DIR}/gif_hash.c ${gif_SOURCE_DIR}/openbsd-reallocarray.c ${gif_SOURCE_DIR}/quantize.c)
target_include_directories(gif_codec PUBLIC ${gif_SOURCE_DIR})
add_library(brushkin_export SHARED export.cpp decode.cpp)
add_dependencies(brushkin_export jpeg_build)
target_include_directories(brushkin_export PRIVATE ${jpeg_SOURCE_DIR}/src ${jpeg_BINARY_DIR} ${png_SOURCE_DIR} ${png_BINARY_DIR} ${webp_SOURCE_DIR}/src)
target_link_libraries(brushkin_export PRIVATE jpeg_codec png_static webp webpdemux libwebpmux gif_codec)
target_compile_features(brushkin_export PRIVATE cxx_std_17)
target_link_options(brushkin_export PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
